LEAKY BUCKET

import java.util.*;

class LeakyBucket {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter bucket size: ");
        int bucketSize = sc.nextInt();

        System.out.print("Enter output rate: ");
        int outputRate = sc.nextInt();

        System.out.print("Enter number of packets: ");
        int numPackets = sc.nextInt();

        int time = 0;

        for (int i = 0; i < numPackets; i++) {
            System.out.print("Enter packet size: ");
            int packetSize = sc.nextInt();

            System.out.println("\nPacket at time " + time + " : " + packetSize);

            // Overflow check
            if (packetSize > bucketSize) {
                System.out.println("Overflow! Packet dropped.");
                continue;
            }

            int remaining = packetSize;

            // Send data step by step
            while (remaining > 0) {
                int sent;

                if (remaining >= outputRate)
                    sent = outputRate;
                else
                    sent = remaining;

                remaining -= sent;
                time++;

                System.out.println("Time " + time + " sent : " + sent + " left : " + remaining);
            }
        }
    }
}

VIGNERE

import java.util.*;

class Vigenere {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter text: ");
        String text = sc.next().toUpperCase();

        System.out.print("Enter key: ");
        String key = sc.next().toUpperCase();

        System.out.print("1.Encrypt 2.Decrypt: ");
        int mode = sc.nextInt();

        String result = "";

        for (int i = 0; i < text.length(); i++) {
            char textChar = text.charAt(i);
            char keyChar = key.charAt(i % key.length());

            int textValue = textChar - 'A';
            int keyValue = keyChar - 'A';

            int outputValue;

            if (mode == 1) { // Encrypt
                outputValue = (textValue + keyValue) % 26;
            } else { // Decrypt
                outputValue = (textValue - keyValue + 26) % 26;
            }

            char outputChar = (char)(outputValue + 'A');
            result += outputChar;
        }

        System.out.println("Result: " + result);
    }
}

DIFFIE HIEMAN

import java.util.*;

class DiffieHellman {

    static int modPower(int base, int exponent, int mod) {
        int result = 1;
        for (int i = 0; i < exponent; i++) {
            result = (result * base) % mod;
        }
        return result;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter prime number p: ");
        int p = sc.nextInt();

        System.out.print("Enter generator g: ");
        int g = sc.nextInt();

        System.out.print("Enter private key of A: ");
        int privateA = sc.nextInt();

        System.out.print("Enter private key of B: ");
        int privateB = sc.nextInt();

        // Public keys
        int publicA = modPower(g, privateA, p);
        int publicB = modPower(g, privateB, p);

        // Shared secret keys
        int sharedKeyA = modPower(publicB, privateA, p);
        int sharedKeyB = modPower(publicA, privateB, p);

        System.out.println("Public key of A: " + publicA);
        System.out.println("Public key of B: " + publicB);
        System.out.println("Shared key (A): " + sharedKeyA);
        System.out.println("Shared key (B): " + sharedKeyB);
    }
}

CRC

import java.util.*;

class CRC {

    // XOR operation
    static String xor(String a, String b) {
        String result = "";
        for (int i = 1; i < b.length(); i++) {
            if (a.charAt(i) == b.charAt(i))
                result += "0";
            else
                result += "1";
        }
        return result;
    }

    // Division (mod-2)
    static String mod2div(String dividend, String divisor) {
        int len = divisor.length();
        String temp = dividend.substring(0, len);

        for (int i = len; i < dividend.length(); i++) {
            if (temp.charAt(0) == '1')
                temp = xor(divisor, temp) + dividend.charAt(i);
            else
                temp = xor("0".repeat(len), temp) + dividend.charAt(i);
        }

        if (temp.charAt(0) == '1')
            temp = xor(divisor, temp);
        else
            temp = xor("0".repeat(len), temp);

        return temp;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter dividend: ");
        String dividend = sc.next();

        String divisor = "10001000000100001";

        // append zeros
        String appended = dividend;
        for (int i = 0; i < divisor.length() - 1; i++) {
            appended += "0";
        }

        String remainder = mod2div(appended, divisor);
        String codeword = dividend + remainder;

        System.out.println("Checksum: " + remainder);
        System.out.println("Codeword: " + codeword);

        // error check
        System.out.print("Check error? (0/1): ");
        int choice = sc.nextInt();

        if (choice == 0) {
            System.out.print("Enter received data: ");
            String received = sc.next();

            String check = mod2div(received, divisor);

            if (check.contains("1"))
                System.out.println("Error detected");
            else
                System.out.println("No error");
        }
    }
}

RSA
import java.util.*;

class RSA {

    static int modPower(int base, int exponent, int mod) {
        int result = 1;
        for (int i = 0; i < exponent; i++) {
            result = (result * base) % mod;
        }
        return result;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter prime numbers p and q: ");
        int p = sc.nextInt();
        int q = sc.nextInt();

        int n = p * q;
        int phi = (p - 1) * (q - 1);

        System.out.print("Enter public key e: ");
        int e = sc.nextInt();

        // find private key d
        int d = 1;
        while ((d * e) % phi != 1) {
            d++;
        }

        System.out.print("Enter message: ");
        int message = sc.nextInt();

        int encrypted = modPower(message, e, n);
        int decrypted = modPower(encrypted, d, n);

        System.out.println("Encrypted: " + encrypted);
        System.out.println("Decrypted: " + decrypted);
    }
}

CAESER

import java.util.*;

class Caesar {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter text: ");
        String text = sc.nextLine();

        System.out.print("Enter key: ");
        int key = sc.nextInt();

        System.out.print("1.Encrypt 2.Decrypt: ");
        int mode = sc.nextInt();

        String result = "";

        if (mode == 2) key = 26 - key;

        for (int i = 0; i < text.length(); i++) {
            char ch = text.charAt(i);

            if (Character.isLetter(ch)) {
                char base;

                if (Character.isUpperCase(ch))
                    base = 'A';
                else
                    base = 'a';

                ch = (char)((ch - base + key) % 26 + base);
            }

            result += ch;
        }

        System.out.println("Result: " + result);
    }
}

PLAYFAIR

import java.util.*;

class Playfair {
    static char[][] mat = new char[5][5];

    static void createMatrix(String key) {
        boolean used[] = new boolean[26];
        key = key.toLowerCase();
        int k = 0;

        for (int i = 0; i < key.length(); i++) {
            char ch = key.charAt(i);
            if (ch == 'j') ch = 'i';

            if (!used[ch - 'a']) {
                mat[k / 5][k % 5] = ch;
                used[ch - 'a'] = true;
                k++;
            }
        }

        for (char ch = 'a'; ch <= 'z'; ch++) {
            if (ch == 'j') continue;

            if (!used[ch - 'a']) {
                mat[k / 5][k % 5] = ch;
                k++;
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter key: ");
        String key = sc.next();

        System.out.print("Enter text: ");
        String text = sc.next();

        System.out.print("1.Encrypt 2.Decrypt: ");
        int mode = sc.nextInt();

        createMatrix(key);

        String result = "";
        int shift = (mode == 1) ? 1 : 4;

        for (int i = 0; i < text.length(); i += 2) {
            char a = text.charAt(i);
            char b = text.charAt(i + 1);

            if (a == 'j') a = 'i';
            if (b == 'j') b = 'i';

            int r1 = 0, c1 = 0, r2 = 0, c2 = 0;

            for (int r = 0; r < 5; r++) {
                for (int c = 0; c < 5; c++) {
                    if (mat[r][c] == a) { r1 = r; c1 = c; }
                    if (mat[r][c] == b) { r2 = r; c2 = c; }
                }
            }

            if (r1 == r2) {
                result += mat[r1][(c1 + shift) % 5];
                result += mat[r2][(c2 + shift) % 5];
            }
            else if (c1 == c2) {
                result += mat[(r1 + shift) % 5][c1];
                result += mat[(r2 + shift) % 5][c2];
            }
            else {
                result += mat[r1][c2];
                result += mat[r2][c1];
            }
        }

        System.out.println("Result: " + result);
    }
}
